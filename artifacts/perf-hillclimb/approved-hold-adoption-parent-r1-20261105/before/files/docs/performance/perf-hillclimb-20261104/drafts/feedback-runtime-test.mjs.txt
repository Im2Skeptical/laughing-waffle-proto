import assert from 'node:assert/strict';
import vm from 'node:vm';
import {installRuntime} from './browser-runtime.mjs';
import {installFeedbackObserver} from './feedback-runtime.mjs';

function fixture({timeoutMs = 1000} = {}) {
  let clock = 100, nextId = 1;
  const frames = new Map(), timers = new Map(), microtasks = [];
  const listeners = new Map();
  const document = {
    addEventListener(type, fn) {
      const list = listeners.get(type) ?? [];
      list.push(fn); listeners.set(type, list);
    },
    removeEventListener(type, fn) {
      listeners.set(type, (listeners.get(type) ?? []).filter(item => item !== fn));
    },
    querySelector: () => null,
  };
  const context = vm.createContext({document,
    performance: {now: () => clock},
    requestAnimationFrame(fn) { const id = nextId++; frames.set(id, fn); return id; },
    cancelAnimationFrame(id) { frames.delete(id); },
    setTimeout(fn, delay = 0) { const id = nextId++; timers.set(id, {fn, at: clock + Math.max(0, delay)}); return id; },
    clearTimeout(id) { timers.delete(id); },
    queueMicrotask(fn) { microtasks.push(fn); },
  });
  const run = source => vm.runInContext(source, context);
  run(`(${installRuntime.toString()})({instrumentation:false})`);
  run(`(${installFeedbackObserver.toString()})()`);
  run("globalThis.cue = {sourceKey:'recorded-action-lineage',sourceValid:true,visible:true,responseKey:'baseline'}");
  const drainMicrotasks = () => {
    let remaining = 1000;
    while (microtasks.length) {
      assert.ok(remaining-- > 0, 'bounded virtual microtask drain');
      microtasks.shift()();
    }
  };
  const drainTimers = () => {
    let remaining = 1000;
    while (true) {
      const due = [...timers].filter(([, t]) => t.at <= clock).sort((a, b) => a[1].at - b[1].at);
      if (!due.length) return;
      assert.ok(remaining-- > 0, 'bounded virtual timer drain');
      const [id, item] = due[0]; timers.delete(id); item.fn(); drainMicrotasks();
    }
  };
  const advance = (ms, {timers: executeTimers = true} = {}) => {
    assert.ok(Number.isFinite(ms) && ms >= 0);
    clock += ms;
    if (executeTimers) drainTimers();
  };
  const frame = (ms = 1000 / 60, options) => {
    advance(ms, options);
    const due = [...frames];
    for (const [id] of due) frames.delete(id);
    for (const [, fn] of due) { fn(clock); drainMicrotasks(); }
  };
  const dispatch = ({trusted = true, timestamp = clock, type = 'pointerdown'} = {}) => {
    const event = {isTrusted: trusted, timeStamp: timestamp, type};
    for (const fn of [...(listeners.get(type) ?? [])]) fn(event);
    drainMicrotasks();
    return event;
  };
  const arm = (reader = '() => cue') => {
    run(`__HC__.arm({name:'feedback-control',timeoutMs:${timeoutMs}})`);
    return run(`__HC_FEEDBACK__.arm({operationId:__HC__.active.id,binding:'virtual-control',readCue:${reader}})`);
  };
  const snapshot = () => JSON.parse(JSON.stringify(run('__HC_FEEDBACK__.snapshot()')));
  const finish = completionAtMs => JSON.parse(JSON.stringify(run(
    completionAtMs === undefined ? '__HC_FEEDBACK__.finish()'
      : `__HC_FEEDBACK__.finish({completionAtMs:${completionAtMs}})`)));
  const assertClean = () => {
    if (run('__HC__.active !== null')) run("__HC__.cancel('virtual fixture cleanup')");
    assert.equal(run('__HC_FEEDBACK__.active'), false, 'feedback collection releases active slot');
    assert.equal(frames.size, 0, 'no observer animation frame survives collection');
    assert.equal(timers.size, 0, 'no runtime/observer timer survives cleanup');
    assert.equal([...listeners.values()].reduce((n, list) => n + list.length, 0), 0, 'no input listener survives cleanup');
    assert.equal(microtasks.length, 0, 'no queued observer microtask survives');
  };
  return {run, arm, snapshot, finish, frame, advance, dispatch, assertClean,
    get now() { return clock; },
    get queuedFrames() { return frames.size; },
    get queuedTimers() { return timers.size; },
    history: () => JSON.parse(JSON.stringify(run('__HC_FEEDBACK__.history()'))),
  };
}

const results = [];
function test(name, body) {
  body(); results.push({name, ok: true});
}
const changed = f => f.run("cue.responseKey='applied-response'");

test('idle installation schedules no observer work', () => {
  const f = fixture();
  assert.equal(f.queuedFrames, 0); assert.equal(f.queuedTimers, 0);
  assert.equal(f.run('__HC_FEEDBACK__.active'), false);
});

test('original trusted timestamp retains queue delay and separate semantic/frame markers', () => {
  const f = fixture(); f.arm(); f.advance(20);
  // Runtime capture handler runs before feedback; app changes the cue after input.
  f.dispatch({timestamp: 105}); changed(f);
  f.frame(); assert.equal(f.snapshot().phase, 'observing');
  f.frame(); assert.equal(f.snapshot().phase, 'observing');
  f.frame();
  const actual = f.finish(f.now);
  assert.equal(actual.phase, 'completed'); assert.equal(actual.startedAtMs, 105);
  assert.equal(actual.inputQueueMs, 15); assert.equal(actual.trustedInput, true);
  assert.ok(actual.firstFeedbackMs > actual.firstResponseSemanticMs);
  assert.equal(actual.firstFeedbackMs, actual.firstFeedbackAtMs - 105);
  assert.equal(actual.comparisonEligible, false); assert.ok(actual.readCount >= 4);
  f.assertClean();
});

test('early feedback precedes the final event of sustained60Hz input', () => {
  const f = fixture(); f.arm(); f.dispatch();
  let first = null;
  for (let index = 1; index <= 14; index++) {
    f.advance(1000 / 60);
    f.run(`cue.responseKey='applied-pose-${index}'`);
    f.dispatch();
    f.frame(0);
    const state = f.snapshot();
    if (!first && state.phase === 'completed') first = state;
  }
  const finalEventAt = f.now, actual = f.finish(finalEventAt);
  assert.ok(first); assert.equal(actual.phase, 'completed');
  assert.ok(actual.firstFeedbackAtMs < finalEventAt);
  assert.equal(f.run('__HC__.active.inputCount'), 15, 'real runtime still receives all offered input');
  assert.equal(f.run('__HC__.active.completionCandidate ?? null'), null, 'feedback does not forge final readiness');
  f.assertClean();
});

test('later heavy completion does not rewrite an earlier feedback marker', () => {
  const f = fixture(); f.arm(); f.dispatch(); changed(f);
  f.frame(); f.frame(); f.frame();
  const before = f.snapshot(); f.advance(400);
  const actual = f.finish(f.now);
  assert.equal(actual.firstFeedbackAtMs, before.firstFeedbackAtMs);
  assert.equal(actual.firstFeedbackMs, before.firstFeedbackMs);
  assert.equal(actual.markerAfterCompletion, false); f.assertClean();
});
