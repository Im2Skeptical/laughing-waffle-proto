// UNAPPLIED, UNEXECUTED DRAFT. Not imported by any harness or product build.
// Intended as a separately installed browser observer after the Life handoff.
// Keep completion's existing input-count invalidation unchanged.
export function installFeedbackObserver() {
  if (globalThis.__HC_FEEDBACK__?.active) {
    throw new Error('cannot replace an active feedback observer');
  }
  const now = () => performance.now();
  let active = null;
  const history = [];
  const copy = value => JSON.parse(JSON.stringify(value));
  const validCue = value => {
    if (!value || typeof value !== 'object' || Array.isArray(value)
      || ![Object.prototype, null].includes(Object.getPrototypeOf(value))) {
      throw new Error('feedback cue must be a plain synchronous DTO');
    }
    const descriptors = Object.getOwnPropertyDescriptors(value);
    const allKeys = Reflect.ownKeys(descriptors);
    if (allKeys.some(key => typeof key !== 'string')) {
      throw new Error('feedback cue symbol fields are forbidden');
    }
    const keys = allKeys.sort();
    const required = ['responseKey', 'sourceKey', 'sourceValid', 'visible'];
    if (keys.length !== required.length || keys.some((key, i) => key !== required[i])) {
      throw new Error('feedback cue requires exactly responseKey/sourceKey/sourceValid/visible');
    }
    for (const key of keys) {
      if (!Object.hasOwn(descriptors[key], 'value') || !descriptors[key].enumerable) {
        throw new Error('feedback cue accessors are forbidden');
      }
    }
    const plain = Object.fromEntries(keys.map(key => [key, descriptors[key].value]));
    if (typeof plain.visible !== 'boolean' || typeof plain.sourceValid !== 'boolean'
      || typeof plain.responseKey !== 'string' || plain.responseKey.length > 4096
      || typeof plain.sourceKey !== 'string' || !plain.sourceKey.length
      || plain.sourceKey.length > 4096) {
      throw new Error('invalid feedback cue primitive fields');
    }
    return plain;
  };
  const describeError = error => error instanceof Error ? error.message : String(error);

  function arm({operationId, binding, readCue}) {
    const runtime = globalThis.__HC__;
    const operation = runtime?.active;
    if (active) throw new Error('overlapping feedback observation');
    if (runtime?.version !== 4 || operation?.id !== operationId) {
      throw new Error('feedback must bind the currently armed runtimev4 operation');
    }
    if (operation.started !== null) throw new Error('feedback must arm before real input');
    if (typeof readCue !== 'function' || typeof binding !== 'string' || !binding.length) {
      throw new Error('named feedback binding and synchronous cue reader required');
    }
    if (!Number.isFinite(operation.timeoutMs) || operation.timeoutMs <= 0
      || !Number.isFinite(operation.armed)) {
      throw new Error('feedback requires a finite host operation deadline');
    }
    let frameId = null, deadlineId = null;
    let candidate = null, finalized = false;
    const state = {
      version: 1, operationId, binding, event: operation.event,
      phase: 'arming', armedAtMs: now(), startedAtMs: null,
      handlerAtMs: null, inputQueueMs: null, trustedInput: null,
      baseline: null, firstResponseSemanticMs: null,
      firstFeedbackAtMs: null, firstFeedbackMs: null,
      frameValidations: 2, observedInputCount: 0,
      readCount: 0, readCpuMs: 0, maxReadCpuMs: 0,
      lastCueReadAtMs: null, lastFrameCueReadAtMs: null,
      lastNegativeFrameCueAtMs: null,
      readSamples: [], readSamplesTruncated: false,
      rejectedCandidates: [], rejectedCandidatesTruncated: false,
      error: null, censor: null, stoppedAtMs: null,
      scope: 'action-related retained visual cue plus double-rAF opportunity; no physical scanout claim',
    };
    const remember = (list, item, truncatedKey) => {
      if (list.length < 512) list.push(item);
      else state[truncatedKey] = true;
    };
    const read = label => {
      const began = now();
      let cue, failure = null;
      try {
        const value = readCue();
        cue = validCue(value);
        return cue;
      } catch (error) {
        failure = describeError(error);
        throw error;
      } finally {
        const cost = now() - began;
        state.readCount++;
        state.readCpuMs += cost;
        state.maxReadCpuMs = Math.max(state.maxReadCpuMs, cost);
        remember(state.readSamples, {atMs: began, cpuMs: cost, label, failure}, 'readSamplesTruncated');
      }
    };
    const clear = () => {
      if (frameId !== null) cancelAnimationFrame(frameId);
      frameId = null;
      clearTimeout(deadlineId);
      deadlineId = null;
      document.removeEventListener(operation.event, onInput, {capture: true});
    };
    const finalize = (phase, reason = null) => {
      if (finalized) return;
      finalized = true;
      state.phase = phase;
      state.stoppedAtMs = now();
      if (phase === 'failed') state.error = reason;
      if (phase === 'censored' || phase === 'cancelled') state.censor = reason;
      clear();
    };
    const rejectCandidate = reason => {
      if (!candidate) return;
      remember(state.rejectedCandidates,
        {semanticAtMs: candidate.semanticAtMs, rejectedAtMs: now(), reason,
          validatedFrames: candidate.validatedFrames, sourceKey: candidate.sourceKey},
        'rejectedCandidatesTruncated');
      candidate = null;
    };
    const stillCurrent = () => {
      if (globalThis.__HC__ !== runtime) return false;
      if (operation.error || operation.timeout) return false;
      if (runtime.active === operation) return true;
      const last = runtime.operations.at(-1);
      return runtime.active === null && last === operation && Number.isFinite(operation.finished);
    };
    const deadline = () => {
      clearTimeout(deadlineId);
      const start = state.startedAtMs ?? operation.armed;
      deadlineId = setTimeout(() => {
        rejectCandidate('feedback observation deadline');
        finalize('censored', state.startedAtMs === null
          ? 'no trusted operation input before deadline'
          : 'no validated visible response before operation deadline');
      }, Math.max(0, start + operation.timeoutMs - now()));
    };
    const observe = (isFrame, label) => {
      if (finalized || state.startedAtMs === null) return;
      if (!stillCurrent()) {
        rejectCandidate('operation cancelled or replaced');
        finalize('cancelled', 'bound operation is no longer current');
        return;
      }
      let cue;
      try { cue = read(label); }
      catch (error) {
        rejectCandidate('cue reader failed');
        finalize('failed', describeError(error));
        return;
      }
      state.lastCueReadAtMs = now();
      if (isFrame) state.lastFrameCueReadAtMs = state.lastCueReadAtMs;
      const changed = cue.sourceValid && cue.visible
        && (!state.baseline.visible || cue.responseKey !== state.baseline.responseKey);
      if (!changed) {
        // A missing/invalid source is unavailable evidence, not visible absence.
        if (isFrame && cue.sourceValid) {
          state.lastNegativeFrameCueAtMs = state.lastFrameCueReadAtMs;
        }
        rejectCandidate(!cue.sourceValid ? 'source correspondence unavailable'
          : !cue.visible ? 'cue is not visible' : 'cue returned to baseline');
        return;
      }
      if (candidate && candidate.sourceKey !== cue.sourceKey) {
        rejectCandidate('cue source lineage changed');
      }
      if (!candidate) {
        candidate = {semanticAtMs: now(), sourceKey: cue.sourceKey,
          responseKey: cue.responseKey, validatedFrames: 0};
        return;
      }
      // Later same-source inputs/changed pose keys do not reset first response.
      // The final-completion observer still owns input-count stabilization.
      if (!isFrame) return;
      candidate.validatedFrames++;
      if (candidate.validatedFrames < state.frameValidations) return;
      state.firstResponseSemanticMs = candidate.semanticAtMs - state.startedAtMs;
      state.firstFeedbackAtMs = now();
      state.firstFeedbackMs = state.firstFeedbackAtMs - state.startedAtMs;
      state.validatedCue = cue;
      if (state.firstFeedbackMs > operation.timeoutMs) {
        rejectCandidate('validated response exceeded deadline');
        state.lateValidatedFeedbackMs = state.firstFeedbackMs;
        state.firstFeedbackMs = null;
        state.firstFeedbackAtMs = null;
        finalize('censored', 'response was only validated after deadline');
      } else finalize('completed');
    };
    const tick = () => {
      frameId = null;
      observe(true, 'animation-frame');
      if (!finalized) frameId = requestAnimationFrame(tick);
    };
    function onInput(event) {
      if (finalized || !event.isTrusted) return;
      state.observedInputCount++;
      if (state.startedAtMs !== null) return;
      const handlerAt = now();
      if (runtime.active !== operation || operation.started === null
        || operation.trustedInput !== true || operation.event !== event.type
        || operation.eventTimeStamp !== event.timeStamp
        || !Number.isFinite(event.timeStamp) || event.timeStamp < 0
        || event.timeStamp > handlerAt + 1) {
        finalize('failed', 'feedback input does not match the operation trusted timestamp');
        return;
      }
      state.startedAtMs = event.timeStamp;
      state.handlerAtMs = handlerAt;
      state.inputQueueMs = handlerAt - event.timeStamp;
      state.trustedInput = true;
      state.phase = 'observing';
      deadline();
      // Run after synchronous app handlers; a cue can still be absent here.
      queueMicrotask(() => observe(false, 'after-input-task'));
      frameId = requestAnimationFrame(tick);
    }
    const controller = {
      state,
      snapshot: () => copy(state),
      cancel(reason) {
        rejectCandidate('observer cancelled');
        finalize('cancelled', String(reason));
        return controller.finish();
      },
      finish({completionAtMs = operation.finished} = {}) {
        if (!finalized) {
          rejectCandidate('completion collected before a validated cue');
          finalize('censored', 'no validated visible response by feedback collection');
        }
        state.completionAtMs = Number.isFinite(completionAtMs) ? completionAtMs : null;
        state.markerAfterCompletion = state.firstFeedbackAtMs !== null && state.completionAtMs !== null
          ? state.firstFeedbackAtMs > state.completionAtMs : null;
        // Never turn host collection/timeout into an unobserved absence bound.
        // This is a negative retained-cue frame sample, still not scanout proof.
        state.censorAtMs = state.phase === 'censored' && state.startedAtMs !== null
          ? state.lastNegativeFrameCueAtMs : null;
        state.censorScope = state.censorAtMs === null
          ? 'no supported negative-frame bound; observation unavailable/incomplete'
          : 'last valid-source negative retained-cue frame sample; no physical absence claim';
        state.comparisonEligible = false;
        state.eligibilityScope = 'construction draft; bindings/controls/overhead/native corroboration unaudited';
        const result = copy(state);
        history.push(result);
        if (active === controller) active = null;
        return result;
      },
    };
    active = controller;
    try {
      state.baseline = read('before-input-baseline');
      if (!state.baseline.sourceValid) throw new Error('feedback baseline source correspondence invalid');
      state.phase = 'waiting-input';
      // Must arm AFTER __HC__.arm so its capture handler binds input first.
      document.addEventListener(operation.event, onInput, {capture: true});
      deadline();
      return operationId;
    } catch (error) {
      finalize('failed', describeError(error));
      controller.finish();
      throw error;
    }
  }
  globalThis.__HC_FEEDBACK__ = {
    version: 1, arm,
    get active() { return active !== null; },
    snapshot: () => active?.snapshot() ?? null,
    finish: options => {
      if (!active) throw new Error('no armed feedback observer');
      return active.finish(options);
    },
    cancel: reason => active?.cancel(reason) ?? null,
    history: () => copy(history),
  };
}
